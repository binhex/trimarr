def validate_cron_expr(expr: str) -> None:
    """Validate a cron expression string.

    Args:
        expr: A standard 5-field cron expression (e.g. ``"0 2 * * *"``)
              or a special string (``@hourly``, ``@daily``, ``@weekly``, ``@monthly``, ``@yearly``).

    Raises:
        ValueError: If the expression cannot be parsed by croniter.
    """
    if not expr or not expr.strip():
        raise ValueError("Cron expression must not be empty.")
    try:
        croniter(expr, datetime.now())
    except (ValueError, KeyError) as exc:
        raise ValueError(f"Invalid cron expression '{expr}': {exc}") from exc
```

def _get_next_fire(cron_expr: str, base: datetime | None = None) -> datetime:
    """Return the next ``datetime`` matching *cron_expr* after *base* (default: now).

    Args:
        cron_expr: Standard 5-field cron expression.
        base: Compute the next match after this point in time. Defaults to
              ``datetime.now()``.

    Returns:
        The next ``datetime`` that matches the cron expression.
    """
    cron = croniter(cron_expr, base or datetime.now())
    return cron.get_next(datetime)


def _sleep_until(target: datetime) -> None:
    """Sleep in 1-second ticks until *target* time, responsive to Ctrl+C.

    Args:
        target: The absolute ``datetime`` to sleep until.
    """
    while True:
        now = datetime.now()
        if now >= target:
            break
        remaining = (target - now).total_seconds()
        if remaining <= 0:
            break
        time.sleep(min(1.0, remaining))
```

```python
def _format_duration(seconds: float) -> str:
    """Convert *seconds* to a compact human-readable string (e.g. '6h 30m', '45m', '0s')."""
    total = int(seconds)
    parts = []
    for unit_char, unit_secs in [
        ("w", 604800),
        ("d", 86400),
        ("h", 3600),
        ("m", 60),
        ("s", 1),
    ]:
        count, total = divmod(total, unit_secs)
        if count:
            parts.append(f"{count}{unit_char}")
    return " ".join(parts) if parts else "0s"


def _sleep_interruptible(seconds: float) -> None:
    """Sleep for *seconds* in 1-second ticks so KeyboardInterrupt is caught promptly.

    Args:
        seconds: Non-negative duration to sleep.
    """
    deadline = time.monotonic() + seconds
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        time.sleep(min(1.0, max(0.0, remaining)))
```

```python
def run_scheduled(
    run_fn: Callable[[], None],
    cron_expr: str,
    run_on_start: bool,
    logger: Logger,
) -> None:
    """Run *run_fn* on a cron-driven schedule until KeyboardInterrupt.

    The next fire time is recomputed from the current clock after each run,
    so any scheduled fires missed during a long-running run are skipped
    automatically.  A warning is logged whenever a fire is missed.

    Args:
        run_fn: Zero-argument callable invoked on each cron tick.
        cron_expr: Standard 5-field cron expression (e.g. ``"0 2 * * *"``)
                   or a special string (``@hourly``, ``@daily``, etc.).
        run_on_start: When ``True``, fire one run immediately before the
                      first cron-driven wait.
        logger: Loguru logger instance for status and warning messages.
    """
    try:
        if not run_on_start:
            next_run = _get_next_fire(cron_expr)
            logger.info(
                f"Scheduler started. First run at "
                f"{next_run.strftime('%Y-%m-%d %H:%M:%S')} "
                f"(in {_format_duration((next_run - datetime.now()).total_seconds())})."
            )
            _sleep_until(next_run)

        while True:
            t0 = time.monotonic()
            try:
                run_fn()
            except SystemExit as exc:
                if exc.code == 130:
                    raise KeyboardInterrupt from exc
                raise
            except Exception as exc:
                logger.error(f"Scheduler: run failed: {exc}")
            elapsed = time.monotonic() - t0

            now = datetime.now()
            next_run = _get_next_fire(cron_expr, now)
            wait_seconds = (next_run - now).total_seconds()

            # Detect missed fires when elapsed exceeds inter-fire gap
            if elapsed > wait_seconds + elapsed:
                logger.warning(
                    f"Scheduler: previous run took {_format_duration(elapsed)}, "
                    f"exceeding the scheduled interval. "
                    f"One or more cron firings were skipped."
                )

            try:
                logger.info(
                    f"Next run at {next_run.strftime('%Y-%m-%d %H:%M:%S')} "
                    f"(in {_format_duration(wait_seconds)})."
                )
            except OverflowError:
                logger.info(f"Next run in {_format_duration(wait_seconds)}.")

            _sleep_until(next_run)
    except KeyboardInterrupt:
        logger.info("Scheduler stopped.")
        sys.exit(0)
```

**Note on overrun detection:** `elapsed > wait_seconds + elapsed` simplifies to `elapsed > elapsed + wait_seconds` which is always `False`. That was a mistake. The correct check should compare `elapsed` against the intended gap between cron firings. Let me fix that:

The intended gap is `(ideal_next - scheduled_fire).total_seconds()`. But since we don't store `scheduled_fire` across loop iterations, the simplest reliable check is: create a croniter before the run, get the *next after that* to get the ideal gap, then compare.
